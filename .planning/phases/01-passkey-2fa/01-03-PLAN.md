**Add passkey sign-in handler:**
```ts
import { startAuthentication } from "@simplewebauthn/browser"

async function handlePasskeySignIn() {
  setError(null)
  setPasskeyLoading(true)

  try {
    // Get authentication options
    const optionsRes = await fetch("/api/auth/passkey/authenticate/options", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ email: email.trim() }),
    })
    if (!optionsRes.ok) throw new Error("Failed to get options")
    const options = await optionsRes.json()

    // Start WebAuthn ceremony
    const authResponse = await startAuthentication({ optionsJSON: options })

    // Verify with server
    const verifyRes = await fetch("/api/auth/passkey/authenticate/verify", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ response: authResponse, email: email.trim() }),
    })
    if (!verifyRes.ok) throw new Error("Verification failed")
    const { passkeyToken } = await verifyRes.json()

    // Sign in with NextAuth using the passkeyToken
    const callbackUrl = searchParams.get("callbackUrl") ?? "/dashboard"
    const result = await signIn("credentials", {
      passkeyToken,
      redirect: false,
    })

    if (result?.error) throw new Error("Sign-in failed")

    // Check if MFA is pending — need to fetch session to check
    const sessionRes = await fetch("/api/auth/session")
    const session = await sessionRes.json()

    if (session?.user?.mfaPending) {
      router.push("/2fa")
    } else {
      router.push(callbackUrl)
    }
  } catch {
    setError("Passkey sign-in failed. Try your password instead.")
  } finally {
    setPasskeyLoading(false)
  }
}
```

**Update `web-app/src/lib/auth.config.ts`** — modify the `authorized` callback to handle mfaPending:

```ts
callbacks: {
  authorized({ auth, request }) {
    const isLoggedIn = !!auth?.user
    if (!isLoggedIn) return false

    // If user has pending MFA, only allow /2fa
    const mfaPending = (auth as { user?: { mfaPending?: boolean } })?.user?.mfaPending
    const isOnMfaPage = request.nextUrl.pathname === "/2fa"

    if (mfaPending && !isOnMfaPage) {
      return Response.redirect(new URL("/2fa", request.nextUrl.origin))
    }

    // If user completed MFA and is on /2fa, send to dashboard
    if (!mfaPending && isOnMfaPage) {
      return Response.redirect(new URL("/dashboard", request.nextUrl.origin))
    }

    return true
  },
},
```

Note: The `authorized` callback in auth.config.ts runs in Edge runtime. It accesses `auth` which is the decoded JWT session — the `mfaPending` field is already in the JWT from Plan 02's auth.ts jwt callback. No Prisma or crypto imports needed here. We access it via type assertion since the Edge config may not have the augmented types directly available.

**Update `web-app/src/middleware.ts`** — add `/2fa` to the matcher so the authorized callback can handle both redirect directions:

```ts
export const config = {
  matcher: [
    "/dashboard/:path*",
    "/courses/:path*",
    "/graphs/:path*",
    "/calendar/:path*",
    "/integrations/:path*",
    "/settings/:path*",
    "/admin/:path*",
    "/billing/:path*",
    "/ai/:path*",
    "/2fa",
  ],
}
```

Adding `/2fa` to the matcher means:
- Unauthenticated users going to /2fa get redirected to /login (by the `if (!isLoggedIn)` check)
- Authenticated users with mfaPending: false going to /2fa get redirected to /dashboard
- Authenticated users with mfaPending: true going to /2fa are allowed through (the authorized callback returns true)
- Authenticated users with mfaPending: true going to ANY other protected route get redirected to /2fa

**Important edge case**: The `authorized` callback receives `auth` which is the session from the JWT. The `mfaPending` field must be present in the JWT token. This was set up in Plan 02 (auth.ts jwt callback writes `token.mfaPending`). The auth.config.ts authorized callback reads from the decoded session, which contains whatever was put in the JWT.

The type assertion `(auth as { user?: { mfaPending?: boolean } })` is needed because the Edge-safe auth.config.ts may not have access to the augmented next-auth types. This is a safe access pattern — if mfaPending is undefined (e.g., for OAuth users without TOTP), it's treated as falsy and they pass through normally.
